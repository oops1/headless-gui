package ui

// Button — кнопка панели инструментов.
type Button struct {
	Text    string
	Pad     int
	Icon    Image
	Enabled bool
}

func New(text string) *Button {
	return &Button{Text: text, Pad: 12, Enabled: true}
}

func (b *Button) Draw(c Ctx) {
	c.Fill(b.Rect, b.Background)
	c.Icon(b.Icon, b.Rect)
	c.Text(b.Text, b.Rect, b.Foreground)
}

func (b *Button) Click() {
	if b.Enabled {
		b.OnClick()
	}
}
