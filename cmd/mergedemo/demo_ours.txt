package ui

// Button — кнопка панели инструментов.
type Button struct {
	Text    string
	Pad     int
	Radius  int
	Enabled bool
}

func New(text string) *Button {
	return &Button{Text: text, Pad: 10, Radius: 6, Enabled: true}
}

func (b *Button) Draw(c Ctx) {
	c.Round(b.Rect, b.Radius, b.Background)
	c.Text(b.Text, b.Rect, b.Foreground)
}

func (b *Button) Click() {
	if b.Enabled {
		b.OnClick()
	}
}
